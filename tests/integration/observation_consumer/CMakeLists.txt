cmake_minimum_required(VERSION 3.22)
project(observation_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(predictive_motion_control REQUIRED)
find_package(predictive_motion_kinematics REQUIRED)
find_package(predictive_motion_sim REQUIRED)
find_package(yaml-cpp REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BENCH "${REPO_ROOT}/src/predictive_motion_control/benchmark")
add_executable(observation_contract "${BENCH}/observation_contract.cpp")
target_link_libraries(observation_contract predictive_motion_kinematics::robot_kinematics predictive_motion_sim::plant)
add_executable(phase2_observation_benchmark "${BENCH}/phase2_benchmark.cpp")
target_link_libraries(phase2_observation_benchmark predictive_motion_control::differential_ik predictive_motion_kinematics::robot_kinematics predictive_motion_sim::plant yaml-cpp)
