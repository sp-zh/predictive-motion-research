cmake_minimum_required(VERSION 3.16)
project(phase5_public_coupled_augmented_cpp LANGUAGES CXX)
add_subdirectory(../phase5_public_coupled_cpp_v2 base-v2)
add_library(public_coupled_augmented_model public_coupled_augmented_model.cpp)
target_include_directories(public_coupled_augmented_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(public_coupled_augmented_model PUBLIC cxx_std_17)
target_compile_options(public_coupled_augmented_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_augmented_model PUBLIC public_coupled_model_v2 PRIVATE yaml-cpp)
add_executable(public_coupled_augmented_probe public_coupled_augmented_probe.cpp)
target_compile_features(public_coupled_augmented_probe PRIVATE cxx_std_17)
target_compile_options(public_coupled_augmented_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_augmented_probe PRIVATE public_coupled_augmented_model yaml-cpp)
