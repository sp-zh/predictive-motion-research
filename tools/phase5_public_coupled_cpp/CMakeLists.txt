cmake_minimum_required(VERSION 3.16)
project(phase5_public_coupled_cpp LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(pinocchio 4.1.0 EXACT CONFIG REQUIRED)
add_library(public_coupled_model public_coupled_model.cpp coupled_friction_box.cpp)
target_include_directories(public_coupled_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(public_coupled_model PUBLIC cxx_std_17)
target_compile_options(public_coupled_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_model PUBLIC Eigen3::Eigen PRIVATE yaml-cpp pinocchio::pinocchio)
add_executable(public_coupled_probe public_coupled_probe.cpp)
target_compile_features(public_coupled_probe PRIVATE cxx_std_17)
target_compile_options(public_coupled_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_probe PRIVATE public_coupled_model yaml-cpp)
