cmake_minimum_required(VERSION 3.16)
project(phase5_public_coupled_finite_trial_cpp LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(pinocchio 4.1.0 EXACT CONFIG REQUIRED)
add_subdirectory(../phase5_public_coupled_augmented_extension_cpp frozen-extension)
add_library(public_coupled_finite_trial public_coupled_finite_trial.cpp)
target_include_directories(public_coupled_finite_trial PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(public_coupled_finite_trial PUBLIC cxx_std_17)
target_compile_options(public_coupled_finite_trial PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_finite_trial PUBLIC public_coupled_augmented_extension PRIVATE yaml-cpp)
add_executable(public_coupled_finite_trial_probe public_coupled_finite_trial_probe.cpp)
target_compile_features(public_coupled_finite_trial_probe PRIVATE cxx_std_17)
target_compile_options(public_coupled_finite_trial_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_finite_trial_probe PRIVATE public_coupled_finite_trial yaml-cpp)
