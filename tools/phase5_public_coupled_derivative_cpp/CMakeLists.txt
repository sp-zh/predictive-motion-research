cmake_minimum_required(VERSION 3.16)
project(phase5_public_coupled_derivative_cpp LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(pinocchio 4.1.0 EXACT CONFIG REQUIRED)
add_subdirectory(../phase5_public_coupled_cpp_v2 base-v2)
add_library(public_coupled_derivative_model public_coupled_derivative_model.cpp)
target_include_directories(public_coupled_derivative_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(public_coupled_derivative_model PUBLIC cxx_std_17)
target_compile_options(public_coupled_derivative_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_derivative_model PUBLIC public_coupled_model_v2 PRIVATE yaml-cpp pinocchio::pinocchio)
add_executable(public_coupled_derivative_probe public_coupled_derivative_probe.cpp)
target_compile_features(public_coupled_derivative_probe PRIVATE cxx_std_17)
target_compile_options(public_coupled_derivative_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_derivative_probe PRIVATE public_coupled_derivative_model yaml-cpp)
