cmake_minimum_required(VERSION 3.16)
project(phase5_public_coupled_augmented_extension_cpp LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(pinocchio 4.1.0 EXACT CONFIG REQUIRED)
add_subdirectory(../phase5_public_coupled_derivative_cpp physical-derivative)
# Compile the frozen original augmented source into a new target; its old
# CMake/source/binary stay unchanged and base-v2 is created exactly once.
add_library(frozen_augmented_extension_value ../phase5_public_coupled_augmented_cpp/public_coupled_augmented_model.cpp)
target_include_directories(frozen_augmented_extension_value PUBLIC ../phase5_public_coupled_augmented_cpp)
target_compile_features(frozen_augmented_extension_value PUBLIC cxx_std_17)
target_compile_options(frozen_augmented_extension_value PRIVATE -Wall -Wextra -Werror)
target_link_libraries(frozen_augmented_extension_value PUBLIC public_coupled_model_v2 PRIVATE yaml-cpp)
add_library(public_coupled_augmented_extension public_coupled_augmented_extension.cpp)
target_include_directories(public_coupled_augmented_extension PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(public_coupled_augmented_extension PUBLIC cxx_std_17)
target_compile_options(public_coupled_augmented_extension PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_augmented_extension PUBLIC frozen_augmented_extension_value public_coupled_derivative_model PRIVATE yaml-cpp)
add_executable(public_coupled_augmented_extension_probe public_coupled_augmented_extension_probe.cpp)
target_compile_features(public_coupled_augmented_extension_probe PRIVATE cxx_std_17)
target_compile_options(public_coupled_augmented_extension_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(public_coupled_augmented_extension_probe PRIVATE public_coupled_augmented_extension yaml-cpp)
