cmake_minimum_required(VERSION 3.22)
project(predictive_motion_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pinocchio REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenGL REQUIRED)
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "$ENV{MUJOCO_ROOT}/include" REQUIRED)
find_library(MUJOCO_LIBRARY mujoco HINTS "$ENV{MUJOCO_ROOT}/lib" REQUIRED)
get_filename_component(MUJOCO_LIB_DIR "${MUJOCO_LIBRARY}" DIRECTORY)
set(CMAKE_INSTALL_RPATH "${MUJOCO_LIB_DIR}")
add_library(plant src/plant.cpp)
target_include_directories(plant PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${MUJOCO_INCLUDE_DIR})
target_link_libraries(plant PUBLIC ${MUJOCO_LIBRARY})
add_executable(state_bridge src/state_bridge.cpp)
target_link_libraries(state_bridge plant)
ament_target_dependencies(state_bridge rclcpp sensor_msgs rosgraph_msgs std_srvs)
add_executable(render_model src/render_model.cpp)
target_link_libraries(render_model plant glfw OpenGL::GL)
add_executable(pinocchio_probe src/pinocchio_probe.cpp)
target_link_libraries(pinocchio_probe pinocchio::pinocchio)
install(TARGETS plant EXPORT plantTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS state_bridge render_model pinocchio_probe DESTINATION lib/${PROJECT_NAME})
ament_export_targets(plantTargets HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(plant_test test/plant_test.cpp)
  target_link_libraries(plant_test plant)
  add_executable(bridge_probe test/bridge_probe.cpp)
  ament_target_dependencies(bridge_probe rclcpp sensor_msgs std_srvs)
  install(TARGETS bridge_probe DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
