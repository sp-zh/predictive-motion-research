cmake_minimum_required(VERSION 3.22)
project(predictive_motion_kinematics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(pinocchio REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(robot_kinematics src/robot_kinematics.cpp src/config.cpp)
target_include_directories(robot_kinematics PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(robot_kinematics PUBLIC pinocchio::pinocchio Eigen3::Eigen yaml-cpp)
install(TARGETS robot_kinematics EXPORT kinematicsTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/fr3.yaml" INSTALLED_CONFIG)
string(REPLACE "../../../models/" "../models/" INSTALLED_CONFIG "${INSTALLED_CONFIG}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fr3.yaml" "${INSTALLED_CONFIG}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fr3.yaml" DESTINATION share/${PROJECT_NAME}/config)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../models/" DESTINATION share/${PROJECT_NAME}/models
        FILES_MATCHING PATTERN "*.urdf" PATTERN "LICENSE" PATTERN "NOTICE" PATTERN "*.commit" PATTERN "*.sha256")
add_executable(kinematics_smoke test/installed_smoke.cpp)
target_link_libraries(kinematics_smoke robot_kinematics)
install(TARGETS kinematics_smoke DESTINATION lib/${PROJECT_NAME})
ament_export_targets(kinematicsTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(pinocchio Eigen3 yaml-cpp)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(kinematics_test test/kinematics_test.cpp)
  target_link_libraries(kinematics_test robot_kinematics)
  find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "$ENV{MUJOCO_ROOT}/include" REQUIRED)
  find_library(MUJOCO_LIBRARY mujoco HINTS "$ENV{MUJOCO_ROOT}/lib" REQUIRED)
  get_filename_component(MUJOCO_LIB_DIR "${MUJOCO_LIBRARY}" DIRECTORY)
  add_executable(numerical_validation test/numerical_validation.cpp)
  target_include_directories(numerical_validation PRIVATE ${MUJOCO_INCLUDE_DIR})
  target_link_libraries(numerical_validation robot_kinematics ${MUJOCO_LIBRARY})
  set_target_properties(numerical_validation PROPERTIES INSTALL_RPATH "${MUJOCO_LIB_DIR}")
  install(TARGETS numerical_validation DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
