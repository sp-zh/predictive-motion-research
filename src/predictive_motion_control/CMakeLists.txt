cmake_minimum_required(VERSION 3.22)
project(predictive_motion_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(differential_ik src/ik.cpp src/nullspace.cpp)
target_include_directories(differential_ik PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(differential_ik PUBLIC Eigen3::Eigen)
install(TARGETS differential_ik EXPORT controlTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(controlTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
option(BUILD_REACTIVE_QP "Build pinned OSQP 1.0 constrained controller" OFF)
if(BUILD_REACTIVE_QP)
  find_package(osqp 1.0.0 EXACT CONFIG REQUIRED)
  add_library(reactive_qp src/reactive_qp.cpp)
  target_include_directories(reactive_qp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(reactive_qp PUBLIC Eigen3::Eigen PRIVATE osqp::osqpstatic)
  install(TARGETS reactive_qp EXPORT qpTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
  add_library(predictive_controller src/predictive.cpp)
  target_include_directories(predictive_controller PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(predictive_controller PUBLIC reactive_qp Eigen3::Eigen)
  install(TARGETS predictive_controller EXPORT qpTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
  ament_export_targets(qpTargets HAS_LIBRARY_TARGET)
  ament_export_dependencies(osqp)
endif()
option(BUILD_BENCHMARKS "Build model-specific validation runner" ON)
if(BUILD_BENCHMARKS)
  find_package(predictive_motion_kinematics REQUIRED)
  find_package(predictive_motion_sim REQUIRED)
  find_package(yaml-cpp REQUIRED)
  add_executable(phase2_benchmark benchmark/phase2_benchmark.cpp)
  target_link_libraries(phase2_benchmark differential_ik predictive_motion_kinematics::robot_kinematics predictive_motion_sim::plant yaml-cpp)
  get_filename_component(MUJOCO_LIB_DIR "$ENV{MUJOCO_ROOT}/lib" ABSOLUTE)
  set_target_properties(phase2_benchmark PROPERTIES INSTALL_RPATH "${MUJOCO_LIB_DIR}")
  install(TARGETS phase2_benchmark DESTINATION lib/${PROJECT_NAME})
  add_executable(phase3_benchmark benchmark/phase3_benchmark.cpp)
  target_link_libraries(phase3_benchmark differential_ik predictive_motion_kinematics::robot_kinematics predictive_motion_sim::plant yaml-cpp)
  set_target_properties(phase3_benchmark PROPERTIES INSTALL_RPATH "${MUJOCO_LIB_DIR}")
  add_executable(phase3_numerical benchmark/phase3_numerical.cpp)
  target_link_libraries(phase3_numerical differential_ik predictive_motion_kinematics::robot_kinematics yaml-cpp)
  install(TARGETS phase3_benchmark phase3_numerical DESTINATION lib/${PROJECT_NAME})
  add_executable(observation_contract benchmark/observation_contract.cpp)
  target_link_libraries(observation_contract differential_ik predictive_motion_kinematics::robot_kinematics predictive_motion_sim::plant)
  set_target_properties(observation_contract PROPERTIES INSTALL_RPATH "${MUJOCO_LIB_DIR}")
  install(TARGETS observation_contract DESTINATION lib/${PROJECT_NAME})
endif()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  if(BUILD_REACTIVE_QP)
    ament_add_gtest(reactive_qp_test test/reactive_qp_test.cpp)
    target_link_libraries(reactive_qp_test reactive_qp)
    ament_add_gtest(predictive_test test/predictive_test.cpp)
    target_link_libraries(predictive_test predictive_controller)
  endif()
  ament_add_gtest(ik_test test/ik_test.cpp)
  target_link_libraries(ik_test differential_ik)
  ament_add_gtest(nullspace_test test/nullspace_test.cpp)
  target_link_libraries(nullspace_test differential_ik)
endif()
ament_package()
